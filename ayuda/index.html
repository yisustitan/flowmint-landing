// Envío del Formulario de Feedback a Brevo async function submitFeedbackForm() { const emailInput = document.getElementById('fbUserEmail'); const valMsg = document.getElementById('fbValidationMsg'); if (!emailInput.value || !emailInput.checkValidity()) { valMsg.textContent = currentLang === 'es' ? '⚠️ Por favor, ingresa un correo electrónico válido.' : '⚠️ Please enter a valid email address.'; valMsg.classList.add('show'); emailInput.focus(); return; } if (!fbSelectedRating) { valMsg.textContent = currentLang === 'es' ? '⚠️ Por favor, califica tu experiencia antes de enviar.' : '⚠️ Please rate your experience before submitting.'; valMsg.classList.add('show'); document.getElementById('fbStarRow').scrollIntoView({ behavior: 'smooth', block: 'center' }); return; } valMsg.classList.remove('show'); const suggestion = document.getElementById('fbSuggestionText').value.trim(); const recChecked = document.querySelector('input[name="fbRecommend"]:checked'); const recVal = recChecked ? recChecked.value : 'yes'; // Obtenemos el texto legible de la recomendación según el idioma const recLabels = { es: { yes: 'Sí, sin duda', maybe: 'Tal vez', no: 'No por ahora' }, en: { yes: 'Yes, definitely', maybe: 'Maybe', no: 'Not right now' } }; const recommendLabel = recLabels[currentLang][recVal] || recVal; // Estado de carga en el botón const btnSubmit = document.getElementById('fbBtnSubmit'); const btnText = document.getElementById('fbBtnText'); const btnSpinner = document.getElementById('fbBtnSpinner'); btnSubmit.disabled = true; btnSubmit.style.opacity = '0.7'; btnText.style.display = 'none'; btnSpinner.style.display = 'inline'; // Envio a webhook con tipo 'rating' y la recomendación incluida const feedbackPayload = { nombre: emailInput.value, email: emailInput.value, mensaje: suggestion || (currentLang === 'en' ? '(no comment)' : '(sin comentario)'), tipo: 'rating', estrellas: fbSelectedRating, recommendation: recommendLabel }; try { const res = await fetch('/api/webhook', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(feedbackPayload) }); const data = await res.json(); showThankYouFeedback(); } catch (err) { console.error('FlowMint Feedback error:', err); showThankYouFeedback(); } finally { btnSubmit.disabled = false; btnSubmit.style.opacity = ''; btnText.style.display = 'inline'; btnSpinner.style.display = 'none'; } }
